八、技术要点
----------------------------------------------------------------------------------------

ECharts配置:
  markPoint: {
    symbol: 'pin',              // 图钉形状
    symbolSize: 50,             // 标记大小
    label: {
      show: true,
      formatter: (params) => `24H最大\n${params.value}万`,
      color: '#fff',
      fontSize: 11,
      fontWeight: 'bold'
    },
    itemStyle: {
      color: '#ef4444'          // 红色
    },
    data: [{
      coord: [maxIndex, maxAmount],
      value: maxAmount.toFixed(2),
      name: '24小时最大值'
    }]
  }

数据计算:
  const max24hAmount = Math.max(...amounts);
  const max24hIndex = amounts.indexOf(max24hAmount);
  const avgAmount = amounts.reduce((a,b) => a+b, 0) / amounts.length;

Tooltip增强:
  formatter: function(params) {
    const amount = params[0].value;
    const isMax = amount === max24hAmount;
    return `<div>
      ${isMax ? '🔥 24小时最大值<br>' : ''}
      时间: ${params[0].name}<br>
      爆仓金额: ${amount}万<br>
      平均值: ${avgAmount.toFixed(2)}万
    </div>`;
  }
